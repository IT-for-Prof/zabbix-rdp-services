> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add best-practice RDP monitoring + a working RDP certificate/NLA/TLS check to Zabbix template 11501, delivered as a standalone project in `/opt/zabbix-rdp-services`.

**Architecture:** A hand-rolled X.224/RDP negotiation in pure Python (stdlib `socket`/`ssl` + `cryptography`) reads the RDP cert, NLA enforcement, and TLS posture; output is JSON consumed by Zabbix dependent items on top of native `net.tcp.service` availability items. **Phase 0 proves every feature against live hosts with a disposable probe before any production code exists.**

```python
    return datetime.now(UTC).replace(microsecond=0).isoformat()

def emit(payload: dict[str, Any] | list[dict[str, Any]]) -> None:
    print(json.dumps(payload, separators=(",", ":"), default=str))
    raise SystemExit(0)                       # always exit 0; Zabbix reads stdout

def error_envelope(error_code: str, message: str, **extra: Any) -> dict[str, Any]:
```

### C4 — Type-safety (mypy `--strict`) — and NO `assert` (ruff S101 / bandit B101)
The corrected forms below were verified end-to-end: **ruff clean, mypy --strict clean, bandit 0 issues.** Asserts are stripped under `python -O` and trip S101/B101 in this `select=["S"]` project — use these instead:
- Annotate every dict return/local as `dict[str, Any]`; `parse_x224_cc`/`negotiate` → `dict[str, int | None]` (bare `dict` → `type-arg`).
- `der = tls.getpeercert(binary_form=True)` is `bytes | None` → `if der is None: raise ValueError("no peer certificate after TLS handshake")` (NOT `assert`).
- `c = tls.cipher(); cipher = c[0] if c else None` (`cipher()` is `tuple | None`).
- Narrow `selected` by bind-then-None-check (mypy won't narrow via an `in (None, …)` test): `selected = cc["value"]` then `if cc["neg_type"] != NEG_TYPE_RSP or selected is None or selected == PROTOCOL_RDP: return {…rdp…}` — afterwards `selected` is `int`, so `selected & PROTOCOL_HYBRID` typechecks.
- SAN: `from typing import cast` → `san = cast(x509.SubjectAlternativeName, ext.value)` before `get_values_for_type` (NOT `assert isinstance`).
- `_first_cn`: `return str(attrs[0].value) if attrs else None` (`value` is `str | bytes`).

### Task 0.3: Write the disposable validation probe

**Files:**
- Create: `scratch/probe_rdp.py`

- [ ] **Step 1: Write the probe** (complete, runnable; throwaway — not the production script)

```python
#!/usr/bin/env python3
"""DISPOSABLE Phase-0 probe. Validates X.224 negotiation + RDP cert/NLA/TLS
against real hosts. Refactored into rdp_check.py only after the gate passes.

Usage:
  python3 scratch/probe_rdp.py <host> [port] [--capture DIR]
"""
from __future__ import annotations
import datetime as dt
import json
import socket
import ssl
import struct
import sys
from pathlib import Path

from cryptography import x509
from cryptography.x509.oid import ExtensionOID, NameOID

PROTOCOL_SSL = 0x00000001
PROTOCOL_HYBRID = 0x00000002
NEG_TYPE_RSP = 0x02
NEG_TYPE_FAILURE = 0x03
FAIL_HYBRID_REQUIRED = 0x05


def build_x224_cr(requested: int) -> bytes:
    neg_req = struct.pack("<BBHI", 0x01, 0x00, 0x0008, requested)  # type,flags,len,protocols
    body = b"\xe0\x00\x00\x00\x00\x00" + neg_req                   # CR+CDT, DST, SRC, class
    x224 = struct.pack("!B", len(body)) + body                     # LI + body
    return struct.pack("!BBH", 0x03, 0x00, 4 + len(x224)) + x224   # TPKT + X.224


def recv_tpkt(sock: socket.socket) -> bytes:
    head = _recvn(sock, 4)
    total = struct.unpack("!H", head[2:4])[0]
    return head + _recvn(sock, total - 4)


def _recvn(sock: socket.socket, n: int) -> bytes:
    buf = b""
    while len(buf) < n:
        chunk = sock.recv(n - len(buf))
        if not chunk:
            raise ConnectionError("peer closed during read")
        buf += chunk
    return buf


def parse_x224_cc(data: bytes) -> dict:
    if len(data) < 5 or data[0] != 0x03:
        raise ValueError("not a TPKT/X.224 response (likely not RDP)")
    li = data[4]
    rest = data[5 + 6 : 4 + 1 + li]  # skip CC(1)+DST(2)+SRC(2)+class(1) = 6 bytes after LI
    if len(rest) < 8:
        return {"neg_type": None, "value": None}
    neg_type = rest[0]
    value = struct.unpack("<I", rest[4:8])[0]
    return {"neg_type": neg_type, "value": value}


def negotiate(sock: socket.socket, requested: int) -> dict:
    sock.sendall(build_x224_cr(requested))
    return parse_x224_cc(recv_tpkt(sock))


def check_nla(host: str, port: int, timeout: float, capture: Path | None) -> bool | None:
    with socket.create_connection((host, port), timeout) as s:
        s.settimeout(timeout)
        s.sendall(build_x224_cr(PROTOCOL_SSL))
        raw = recv_tpkt(s)
        if capture:
            (capture / f"{host}_nla_ssl_only.bin").write_bytes(raw)
        cc = parse_x224_cc(raw)
    if cc["neg_type"] == NEG_TYPE_FAILURE and cc["value"] == FAIL_HYBRID_REQUIRED:
        return True
    if cc["neg_type"] == NEG_TYPE_RSP:
        return False
    return None  # legacy/standard or unknown


def grab_cert(host: str, port: int, timeout: float, capture: Path | None) -> dict:
    with socket.create_connection((host, port), timeout) as s:
        s.settimeout(timeout)
        cc = negotiate(s, PROTOCOL_HYBRID | PROTOCOL_SSL)
        selected = cc.get("value")
        if cc["neg_type"] != NEG_TYPE_RSP or selected in (None, 0x0):
            return {"security_layer": "rdp", "cert": None, "neg": cc}
        ctx = ssl._create_unverified_context()  # noqa: S323 — probe only
        ctx.check_hostname = False
        ctx.verify_mode = ssl.CERT_NONE
        with ctx.wrap_socket(s, server_hostname=host) as tls:
            der = tls.getpeercert(binary_form=True)
            version, cipher = tls.version(), tls.cipher()[0]
        if capture:
            (capture / f"{host}_cert.der").write_bytes(der)
    cert = x509.load_der_x509_certificate(der)
    layer = "hybrid" if selected & PROTOCOL_HYBRID else "ssl"
    return {"security_layer": layer, "tls": {"version": version, "cipher": cipher},
            "cert": _describe_cert(cert, host)}


def _describe_cert(cert: x509.Certificate, host: str) -> dict:
    not_after = cert.not_valid_after_utc
    cn = _first_cn(cert.subject)
    issuer_cn = _first_cn(cert.issuer)
    names = _san_dns(cert)
    if cn:
        names.append(cn)
    return {
        "subject_cn": cn, "issuer": issuer_cn,
        "not_before": cert.not_valid_before_utc.isoformat(),
        "not_after": not_after.isoformat(),
        "days_to_expiry": (not_after - dt.datetime.now(dt.UTC)).days,
        "self_signed": cert.subject == cert.issuer,
        "hostname_match": _host_in(host, names),
    }


def _first_cn(name: x509.Name) -> str | None:
    attrs = name.get_attributes_for_oid(NameOID.COMMON_NAME)
    return attrs[0].value if attrs else None


def _san_dns(cert: x509.Certificate) -> list[str]:
    try:
        ext = cert.extensions.get_extension_for_oid(ExtensionOID.SUBJECT_ALTERNATIVE_NAME)
        return list(ext.value.get_values_for_type(x509.DNSName))
    except x509.ExtensionNotFound:
        return []


def _host_in(host: str, names: list[str]) -> bool:
    h = host.lower().rstrip(".")
    return any(h == n.lower().rstrip(".") or
               (n.startswith("*.") and h.endswith(n[1:].lower())) for n in names)


def main() -> int:
    host = sys.argv[1]
    port = int(sys.argv[2]) if len(sys.argv) > 2 and sys.argv[2].isdigit() else 3389
    capture = None
    if "--capture" in sys.argv:
        capture = Path(sys.argv[sys.argv.index("--capture") + 1]); capture.mkdir(parents=True, exist_ok=True)
    out: dict = {"host": host, "port": port}
    try:
        out["nla_enforced"] = check_nla(host, port, 10.0, capture)
        out.update(grab_cert(host, port, 10.0, capture))
    except Exception as exc:  # probe: report, don't crash
        out["error"] = f"{type(exc).__name__}: {exc}"
    print(json.dumps(out, indent=2))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
import socket


def _recvn(sock: socket.socket, n: int) -> bytes:
    buf = b""
    while len(buf) < n:
        chunk = sock.recv(n - len(buf))
        if not chunk:
            raise ConnectionError("peer closed during read")
        buf += chunk
    return buf


def recv_tpkt(sock: socket.socket) -> bytes:
    head = _recvn(sock, 4)
    if head[0] != 0x03:
        raise ValueError("not a TPKT response (likely not RDP)")
    total = struct.unpack("!H", head[2:4])[0]
    return head + _recvn(sock, total - 4)


def parse_x224_cc(data: bytes) -> dict:
    if len(data) < 5 or data[0] != 0x03:
        raise ValueError("not a TPKT/X.224 response (likely not RDP)")
    li = data[4]
    rest = data[5 + 6 : 4 + 1 + li]   # skip CC(1)+DST(2)+SRC(2)+class(1)
    if len(rest) < 8:
        return {"neg_type": None, "value": None}
    return {"neg_type": rest[0], "value": struct.unpack("<I", rest[4:8])[0]}


def negotiate(sock: socket.socket, requested_protocols: int) -> dict:
    sock.sendall(build_x224_cr(requested_protocols))
    return parse_x224_cc(recv_tpkt(sock))


def check_nla(host: str, port: int, timeout: float = 10.0) -> bool | None:
    with socket.create_connection((host, port), timeout) as s:
        s.settimeout(timeout)
        cc = negotiate(s, PROTOCOL_SSL)
    if cc["neg_type"] == NEG_TYPE_FAILURE and cc["value"] == FAIL_HYBRID_REQUIRED:
        return True
    if cc["neg_type"] == NEG_TYPE_RSP:
        return False
    return None
```

```python
import datetime as dt
import ssl

from cryptography import x509
from cryptography.x509.oid import ExtensionOID, NameOID


def grab_cert(host: str, port: int, hostname: str, timeout: float = 10.0) -> dict:
    with socket.create_connection((host, port), timeout) as s:
        s.settimeout(timeout)
        cc = negotiate(s, PROTOCOL_HYBRID | PROTOCOL_SSL)
        selected = cc.get("value")
        if cc["neg_type"] != NEG_TYPE_RSP or selected in (None, PROTOCOL_RDP):
            return {"security_layer": "rdp", "tls": None, "cert": None}
        ctx = ssl._create_unverified_context()  # noqa: S323 — cert read, not a trust decision
        with ctx.wrap_socket(s, server_hostname=host) as tls:
            der = tls.getpeercert(binary_form=True)
            version, cipher = tls.version(), tls.cipher()[0]
    cert = x509.load_der_x509_certificate(der)
    layer = "hybrid" if selected & PROTOCOL_HYBRID else "ssl"
    return {"security_layer": layer,
            "tls": {"version": version, "cipher": cipher},
            "cert": _describe_cert(cert, hostname)}


def _describe_cert(cert: x509.Certificate, hostname: str) -> dict:
    not_after = cert.not_valid_after_utc
    cn = _first_cn(cert.subject)
    names = _san_dns(cert) + ([cn] if cn else [])
    return {
        "subject_cn": cn,
        "issuer": _first_cn(cert.issuer),
        "not_before": cert.not_valid_before_utc.isoformat(),
        "not_after": not_after.isoformat(),
        "days_to_expiry": (not_after - dt.datetime.now(dt.UTC)).days,
        "self_signed": cert.subject == cert.issuer,
        "hostname_match": _host_in(hostname, names),
        "chain_valid": False,  # set by run_cert via a verifying handshake (Task 1.5)
    }


def _first_cn(name: x509.Name) -> str | None:
    attrs = name.get_attributes_for_oid(NameOID.COMMON_NAME)
    return attrs[0].value if attrs else None


def _san_dns(cert: x509.Certificate) -> list[str]:
    try:
        ext = cert.extensions.get_extension_for_oid(ExtensionOID.SUBJECT_ALTERNATIVE_NAME)
        return list(ext.value.get_values_for_type(x509.DNSName))
    except x509.ExtensionNotFound:
        return []


def _host_in(host: str, names: list[str]) -> bool:
    h = host.lower().rstrip(".")
    return any(h == n.lower().rstrip(".") or
               (n.startswith("*.") and h.endswith(n[1:].lower())) for n in names)
```

```python
def _chain_validates(host: str, port: int, hostname: str, timeout: float) -> bool:
```

```python
def _try_tls_version(host: str, port: int, ver: ssl.TLSVersion, timeout: float) -> bool:
    try:
        with socket.create_connection((host, port), timeout) as s:
            s.settimeout(timeout)
            cc = negotiate(s, PROTOCOL_HYBRID | PROTOCOL_SSL)
            if cc["neg_type"] != NEG_TYPE_RSP or cc.get("value") in (None, PROTOCOL_RDP):
                return False
            ctx = ssl._create_unverified_context()  # noqa: S323 — posture probe
            ctx.minimum_version = ver
            ctx.maximum_version = ver
            with ctx.wrap_socket(s, server_hostname=host):
                return True
    except (ssl.SSLError, OSError, ValueError):
        return False


def run_tls_scan(host: str, port: int, timeout: float = 10.0) -> dict:
    matrix = {
        "TLSv1.0": ssl.TLSVersion.TLSv1, "TLSv1.1": ssl.TLSVersion.TLSv1_1,
        "TLSv1.2": ssl.TLSVersion.TLSv1_2, "TLSv1.3": ssl.TLSVersion.TLSv1_3,
    }
    offered = [name for name, ver in matrix.items() if _try_tls_version(host, port, ver, timeout)]
    weak = [n for n in offered if n in ("TLSv1.0", "TLSv1.1")]
    return {"host": host, "port": port, "offered": offered,
            "negotiated": offered[-1] if offered else None,
            "weak_protocols": weak, "ts": dt.datetime.now(dt.UTC).isoformat()}
```
